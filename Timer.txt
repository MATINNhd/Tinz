-- ==========================================
-- EXECUTOR SCRIPT WITH CONFIG, PING & FPS (FIXED BUTTONS)
-- ==========================================

local RunService = game:GetService("RunService")
local CoreGui = game:GetService("CoreGui")
local Players = game:GetService("Players")
local VirtualUser = game:GetService("VirtualUser")
local LocalPlayer = Players.LocalPlayer

-- ------------------------------------------
-- 0. ระบบจัดการไฟล์บันทึกค่า (Config System)
-- ------------------------------------------
local ConfigFileName = "AFK_Script_Config.json"

local function saveConfig(isBlackState, isWalkState)
    if writefile then
        local data = {
            BlackScreen = isBlackState,
            AutoWalk = isWalkState
        }
        pcall(function()
            writefile(ConfigFileName, game:GetService("HttpService"):JSONEncode(data))
        end)
    end
end

local function loadConfig()
    if isfile and isfile(ConfigFileName) and readfile then
        local success, result = pcall(function()
            return game:GetService("HttpService"):JSONDecode(readfile(ConfigFileName))
        end)
        if success and type(result) == "table" then
            return result
        end
    end
    return { BlackScreen = false, AutoWalk = false }
end

local savedConfig = loadConfig()

-- ------------------------------------------
-- 1. ประกาศตัวแปรสถานะหลัก
-- ------------------------------------------
local isBlack = savedConfig.BlackScreen
local isWalking = savedConfig.AutoWalk

-- ------------------------------------------
-- 2. ระบบ Anti-AFK
-- ------------------------------------------
LocalPlayer.Idled:Connect(function()
    VirtualUser:CaptureController()
    VirtualUser:ClickButton2(Vector2.new(0, 0))
end)

-- ------------------------------------------
-- 3. สร้าง ScreenGui หลักใน CoreGui
-- ------------------------------------------
local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "Executor_AFK_Suite"
ScreenGui.ResetOnSpawn = false
ScreenGui.IgnoreGuiInset = true

pcall(function() ScreenGui.Parent = CoreGui end)
if not ScreenGui.Parent then ScreenGui.Parent = LocalPlayer:WaitForChild("PlayerGui") end

-- ------------------------------------------
-- 4. UI จอดำเต็มจอ
-- ------------------------------------------
local BlackFrame = Instance.new("Frame")
BlackFrame.Name = "BlackFrame"
BlackFrame.Size = UDim2.new(1, 0, 1, 0)
BlackFrame.Position = UDim2.new(0, 0, 0, 0)
BlackFrame.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
BlackFrame.BorderSizePixel = 0
BlackFrame.Visible = isBlack
BlackFrame.ZIndex = 9998 -- อยู่ใต้ปุ่ม (ปุ่ม ZIndex สูงกว่า จะได้ไม่หาย)
BlackFrame.Parent = ScreenGui

-- ข้อความสถานะกลางจอ (แสดงสถานะ + ปิง + FPS)
local StatusText = Instance.new("TextLabel")
StatusText.Size = UDim2.new(1, 0, 0, 100)
StatusText.Position = UDim2.new(0, 0, 0.5, -50)
StatusText.BackgroundTransparency = 1
StatusText.Text = "AFK MODE ACTIVE\n[Black Screen: ON | Anti-AFK: ON]\nPing: 0 ms | FPS: 0"
StatusText.TextColor3 = Color3.fromRGB(0, 255, 127)
StatusText.TextSize = 20
StatusText.Font = Enum.Font.SourceSansBold
StatusText.ZIndex = 9999
StatusText.Parent = BlackFrame

-- ------------------------------------------
-- 5. แสดงผล Ping และ FPS แบบเรียลไทม์
-- ------------------------------------------
local lastTick = tick()
local frameCount = 0
local currentFPS = 0

RunService.RenderStepped:Connect(function()
    frameCount = frameCount + 1
    if tick() - lastTick >= 1 then
        currentFPS = math.floor(frameCount / (tick() - lastTick))
        frameCount = 0
        lastTick = tick()
    end
    
    pcall(function()
        local ping = math.floor(LocalPlayer:GetNetworkPing() * 1000)
        
        if isBlack then
            StatusText.Text = string.format(
                "AFK MODE ACTIVE\n[Black Screen: ON | Anti-AFK: ON]\nPing: %d ms | FPS: %d", 
                ping, currentFPS
            )
        end
    end)
end)

-- ------------------------------------------
-- 6. ปุ่ม Toggle หน้าจอดำ (กำหนด ZIndex สูงสุดเพื่อให้ปุ่มไม่หาย)
-- ------------------------------------------
local BlackBtn = Instance.new("TextButton")
BlackBtn.Name = "BlackBtn"
BlackBtn.Size = UDim2.new(0, 150, 0, 45)
BlackBtn.Position = UDim2.new(0, 20, 0, 20)
BlackBtn.BackgroundColor3 = isBlack and Color3.fromRGB(0, 170, 0) or Color3.fromRGB(40, 40, 40)
BlackBtn.Text = isBlack and "Black Screen: ON" or "Black Screen: OFF"
BlackBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
BlackBtn.Font = Enum.Font.SourceSansBold
BlackBtn.TextSize = 14
BlackBtn.ZIndex = 10000 -- อยู่เหนือสุด ปุ่มไม่หายแน่นอน
BlackBtn.Parent = ScreenGui

BlackBtn.MouseButton1Click:Connect(function()
    isBlack = not isBlack
    BlackFrame.Visible = isBlack
    
    if isBlack then
        BlackBtn.Text = "Black Screen: ON"
        BlackBtn.BackgroundColor3 = Color3.fromRGB(0, 170, 0)
    else
        BlackBtn.Text = "Black Screen: OFF"
        BlackBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
    end
    saveConfig(isBlack, isWalking)
end)

-- ------------------------------------------
-- 7. ปุ่ม Toggle เดินวน
-- ------------------------------------------
local WalkBtn = Instance.new("TextButton")
WalkBtn.Name = "WalkBtn"
WalkBtn.Size = UDim2.new(0, 150, 0, 45)
WalkBtn.Position = UDim2.new(0, 180, 0, 20)
WalkBtn.BackgroundColor3 = isWalking and Color3.fromRGB(0, 170, 0) or Color3.fromRGB(40, 40, 40)
WalkBtn.Text = isWalking and "Auto Walk: ON" or "Auto Walk: OFF"
WalkBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
WalkBtn.Font = Enum.Font.SourceSansBold
WalkBtn.TextSize = 14
WalkBtn.ZIndex = 10000 -- อยู่เหนือสุด ปุ่มไม่หายแน่นอน
WalkBtn.Parent = ScreenGui

WalkBtn.MouseButton1Click:Connect(function()
    isWalking = not isWalking
    if isWalking then
        WalkBtn.Text = "Auto Walk: ON"
        WalkBtn.BackgroundColor3 = Color3.fromRGB(0, 170, 0)
    else
        WalkBtn.Text = "Auto Walk: OFF"
        WalkBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
        
        pcall(function()
            local char = LocalPlayer.Character
            if char and char:FindFirstChild("Humanoid") then
                char.Humanoid:MoveTo(char.HumanoidRootPart.Position)
            end
        end)
    end
    saveConfig(isBlack, isWalking)
end)

-- ------------------------------------------
-- 8. ลูปการทำงานเดินวน (Auto Walk Loop)
-- ------------------------------------------
task.spawn(function()
    while true do
        task.wait(1)
        if isWalking then
            pcall(function()
                local character = LocalPlayer.Character
                if character and character:FindFirstChild("HumanoidRootPart") and character:FindFirstChild("Humanoid") then
                    local humanoid = character.Humanoid
                    local hrp = character.HumanoidRootPart
                    
                    local startPos = hrp.Position
                    local targetPos = startPos + (hrp.CFrame.LookVector * 15)
                    
                    humanoid:MoveTo(targetPos)
                    humanoid.MoveToFinished:Wait()
                    
                    if not isWalking then return end
                    
                    humanoid:MoveTo(startPos)
                    humanoid.MoveToFinished:Wait()
                end
            end)
        end
    end
end)
